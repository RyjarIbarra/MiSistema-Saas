CREATE TABLE IF NOT EXISTS public.timbrado (
    timid        SERIAL,
    timtipdoc    VARCHAR(3)  NOT NULL,
    timnumero    VARCHAR(8)  NOT NULL,
    timmodalid   CHAR(1)     NOT NULL DEFAULT 'P',
    timfecini    DATE        NOT NULL,
    timfecvto    DATE,
    timestab     CHAR(3)     NOT NULL,
    timpunexp    CHAR(3)     NOT NULL,
    timnrodesde  INTEGER,
    timnrohasta  INTEGER,
    timnroactual INTEGER     NOT NULL,
    timactivo    BOOLEAN     NOT NULL DEFAULT TRUE,
    CONSTRAINT timbrado_pkey   PRIMARY KEY (timid),
    CONSTRAINT timbrado_tipdoc FOREIGN KEY (timtipdoc) REFERENCES public.tipo_documento (tdocodigo),
    CONSTRAINT timbrado_unique UNIQUE (timtipdoc, timnumero, timestab, timpunexp),
    CONSTRAINT timbrado_modalid CHECK (timmodalid IN ('P', 'A', 'E')),
    CONSTRAINT timbrado_fechas  CHECK (timfecvto IS NULL OR timfecvto >= timfecini),
    CONSTRAINT timbrado_rango   CHECK (
        CASE WHEN timmodalid = 'E' THEN
            timnroactual BETWEEN 1 AND 9999999
        ELSE
            timnrodesde IS NOT NULL AND timnrohasta IS NOT NULL
            AND timnrodesde BETWEEN 1 AND 9999999
            AND timnrohasta BETWEEN timnrodesde AND 9999999
            AND timnroactual BETWEEN timnrodesde AND timnrohasta + 1
        END
    )
);
CREATE UNIQUE INDEX IF NOT EXISTS timbrado_vigente_uq
    ON public.timbrado (timtipdoc, timestab, timpunexp, timmodalid)
    WHERE timactivo;

-- =====================================================
-- FUNCIÓN: reserva del próximo número correlativo
-- =====================================================
CREATE OR REPLACE FUNCTION public.fn_timbrado_siguiente_numero(p_timid INTEGER)
RETURNS INTEGER
LANGUAGE plpgsql
AS $$
DECLARE
    v_numero INTEGER;
BEGIN
    UPDATE public.timbrado
       SET timnroactual = timnroactual + 1
     WHERE timid = p_timid
       AND timactivo
       AND (timfecvto IS NULL OR timfecvto >= CURRENT_DATE)
    RETURNING timnroactual - 1 INTO v_numero;

    IF v_numero IS NULL THEN
        RAISE EXCEPTION 'Timbrado % inexistente, inactivo o vencido', p_timid
            USING ERRCODE = 'check_violation';
    END IF;

    RETURN v_numero;
END;
$$;

-------------*** Crea la funcion fn_actualizar_stock_v1() ***-------------

CREATE OR REPLACE FUNCTION public.fn_actualizar_stock_v1()
    RETURNS trigger
    LANGUAGE 'plpgsql'
    COST 100
    VOLATILE NOT LEAKPROOF
AS $BODY$
DECLARE
    v_permite_negativo boolean;
    v_stock_actual numeric(12,3);
BEGIN

	    IF NEW.tipo_movimiento = 'ENTRADA' THEN
	        -- Sumar stock (UPSERT)
	        INSERT INTO stock (id_producto, id_deposito, cantidad)
	        VALUES (NEW.id_producto, NEW.id_deposito, NEW.cantidad)
	        ON CONFLICT (id_producto, id_deposito)
	        DO UPDATE SET cantidad = stock.cantidad + NEW.cantidad;

	    ELSIF NEW.tipo_movimiento = 'SALIDA' THEN

	        -- 1. Consultar la configuración del producto y el stock actual
	        -- Nota: Asumo que tu tabla de productos se llama 'producto' según tu descripción
	        SELECT stocknegativo INTO v_permite_negativo
	        FROM public.producto
	        WHERE proid = NEW.id_producto; -- Ajusta el nombre del campo ID si es diferente (ej: prodid)

	        -- 2. Obtener el stock actual disponible en ese depósito
	        SELECT cantidad INTO v_stock_actual
	        FROM public.stock
	        WHERE id_producto = NEW.id_producto AND id_deposito = NEW.id_deposito;

	        -- Si no existe el registro en la tabla stock, tratamos el stock actual como 0
	        IF v_stock_actual IS NULL THEN
	            v_stock_actual := 0;
	        END IF;

	        -- 3. Validar si se permite stock negativo
	        -- Si NO permite negativo (FALSE) y la resta daría menor a cero, lanzamos error
	        IF NOT v_permite_negativo AND (v_stock_actual - NEW.cantidad) < 0 THEN
	            RAISE EXCEPTION 'Stock insuficiente para el producto (ID: %). Stock actual: %, Solicitado: %. El producto no permite stock negativo.',
	            NEW.id_producto, v_stock_actual, NEW.cantidad;
	        END IF;

	        -- 4. Si pasó la validación o si permite negativos, procedemos a restar
	        UPDATE stock
	        SET cantidad = cantidad - NEW.cantidad
	        WHERE id_producto = NEW.id_producto
	          AND id_deposito = NEW.id_deposito;

	        -- Validación de existencia del registro en la tabla stock
	        IF NOT FOUND THEN
	             -- Si no existe registro y permite negativo, deberíamos insertarlo con valor negativo
	             -- IF v_permite_negativo THEN
	                -- INSERT INTO stock (id_producto, id_deposito, cantidad)

-------------*** Crea Vista v_stock_detallado ***-------------

CREATE VIEW public.v_stock_detallado AS
SELECT
    p.proid,
    p.prodesc AS producto,
    CASE
        WHEN u.descripcion = 'Servicio' THEN 'N/A'::varchar(250)
        ELSE COALESCE(d.depid::text || ' - ' || d.depnom, 'Sin Depósito')::varchar(250)
    END AS deposito,
    COALESCE(s.cantidad, 0) AS cantidad,
    u.descripcion AS unidad,
    CASE
        WHEN u.descripcion = 'Servicio' THEN 'N/A (Servicio)'
        ELSE concat(
            trim(trailing '.' from trim(trailing '0' from COALESCE(s.cantidad, 0)::text)),
            ' ',
            u.abreviatura
        )
    END AS stock_formateado
FROM producto p
JOIN unidad_medida u ON p.unidad = u.codigo
LEFT JOIN stock s ON p.proid = s.id_producto
LEFT JOIN deposito d ON s.id_deposito = d.depid
WHERE s.id_producto IS NOT NULL OR u.descripcion = 'Servicio'
order by p.prodesc;

        (tipo_documento = 9 AND RIGHT(TRIM(prvruc), 2) LIKE '-_')

    )
